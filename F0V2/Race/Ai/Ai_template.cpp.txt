
#include "Asset.generated.h"
#include "TY/ActorContainer.h"
#include "TY/ModelDrawer.h"
#include "TY_Extension/GameObjectBase.h"

using namespace Race;

namespace
{
}

struct Ai_template::Impl : GameObjectBase
{
    ActorContainer m_children{};

    ModelDrawer m_drawer{};

    void Init()
    {
        m_drawer =
            ModelDrawerParams{}
            .setModel(Asset_model::cinnamon)
            .setShader(Asset_shader::model);
    }

private:
    void update() override
    {
        m_drawer.uploadWorldMatrix(Mat4x4::Identity()).draw();
    }

    void killed() override
    {
        m_children.killEach();
    }

    std::u32string name() const override
    {
        return U"Ai_template";
    }
};

namespace Race
{
    Ai_template::Ai_template() :
        p_impl(std::make_shared<Impl>())
    {
    }

    void Ai_template::init()
    {
        p_impl->Init();
        GameObjectHandle::init();
    }

    std::shared_ptr<GameObjectBase> Ai_template::asGameObject() const
    {
        return p_impl;
    }
}
