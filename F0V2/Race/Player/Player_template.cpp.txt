
#include "Asset.generated.h"
#include "TY/ActorContainer.h"

using namespace Race;

namespace
{
}

struct Player_template::Impl : ActorBase
{
#if defined(_DEBUG)
    std::u32string m_debugName = U"Player_template";
#endif
    ActorContainer m_children{};

    void Init()
    {
    }

private:
    void update() override
    {
        m_children.updateEach();
    }

    void killed() override
    {
        m_children.killEach();
    }
};

namespace Race
{
    Player_template::Player_template() :
        p_impl(std::make_shared<Impl>())
    {
    }

    void Player_template::init()
    {
        p_impl->Init();
    }

    std::shared_ptr<ActorBase> Player_template::asActor() const
    {
        return p_impl;
    }
}
